import { NextRequest, NextResponse } from "next/server";

function decodeJWT(token: string): Record<string, unknown> | null {
  try {
    const parts = token.split(".");
    if (parts.length !== 3) return null;

    const payload = JSON.parse(
      atob(parts[1].replace(/-/g, "+").replace(/_/g, "/"))
    );

    if (payload.exp && payload.exp < Math.floor(Date.now() / 1000)) {
      return null;
    }

    return payload;
  } catch {
    return null;
  }
}

export function middleware(request: NextRequest) {
  const authToken = request.cookies.get("authToken")?.value;
  const payload = authToken ? decodeJWT(authToken) : null;

  if (!payload) {
    return NextResponse.redirect(new URL("/login", request.url));
  }

  if (payload.role !== "ADMIN") {
    return NextResponse.redirect(new URL("/login", request.url));
  }

  return NextResponse.next();
}

    13	function timingSafeEqualStrings(a: string, b: string): boolean {
    14	  const length = Math.max(a.length, b.length);
    15	  let diff = a.length ^ b.length;
    16	  for (let i = 0; i < length; i++) {
    17	    const ac = i < a.length ? a.charCodeAt(i) : 0;
    18	    const bc = i < b.length ? b.charCodeAt(i) : 0;

Véase `app/api/orders/route.ts:39`, especialmente las líneas 77 y 130. | Calcular el total con precios, cantidades y descuentos del servidor. Crear pedido y líneas, consumir el cupón con condición atómica y vaciar el carrito en una transacción idempotente. |
| A13 · Tarjetas regalo | La emisión crea valor canjeable sin verificar un pago en la ruta. Los códigos proceden de un generador determinista basado en la hora. El canje marca la tarjeta y después acredita saldo en otra operación, por lo que un fallo intermedio puede consumirla sin abonar el crédito. Véanse `app/api/gift-cards/route.ts:51`, `lib/gift-card.ts:5` y `app/api/gift-cards/redeem/route.ts:52`. | Emitir tras confirmar el pago con referencia única, generar códigos criptográficamente aleatorios y actualizar tarjeta y saldo en una sola transacción. |
| A14 · Enlaces compartidos | Los tokens documentales usan AES-CBC sin autenticación con clave predeterminada y sin caducidad. La ruta pública usa el contenido descifrado para seleccionar el recurso y devolver datos del pedido. Véanse `lib/share-crypto.ts:3` y `app/api/documents/share/route.ts:13`. | Usar identificadores aleatorios de alta entropía asociados en servidor a un solo recurso, con expiración y revocación. Invalidar los enlaces anteriores. |
| A15 · Contraseñas almacenadas | La aplicación usa MD5 sin sal para contraseñas. Una copia de la base permite comprobar con rapidez conjeturas sobre muchas cuentas. El exportador permite además incluir el hash de la propia cuenta. Véanse `lib/server-auth.ts:14`, `app/api/auth/signup/route.ts:46` y `app/api/user/export/route.ts:7`. | Migrar a Argon2id con sal individual y coste calibrado, renovar hashes al autenticar y exigir restablecimiento para cuentas expuestas. Eliminar el hash de los campos exportables. |
